    ⚠️ Passos para Configurar

    1. Criar projeto em supabase.com
    2. Executar o SQL abaixo no editor do Supabase
    3. Copiar URL e chave anon do projeto
    4. Colar no código (linha ~174)

    💾 SQL para Criar Tabelas:

    -- Tabela de cirurgiões
CREATE TABLE cirurgioes (
  id BIGSERIAL PRIMARY KEY,
  nome TEXT NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW())
);

-- Tabela de cirurgias
CREATE TABLE cirurgias (
  id BIGSERIAL PRIMARY KEY,
  paciente TEXT NOT NULL,
  data DATE NOT NULL,
  numero_mes INTEGER NOT NULL,
  cirurgiao_id BIGINT REFERENCES cirurgioes(id) ON DELETE SET NULL,
  auxiliar1_id BIGINT REFERENCES cirurgioes(id) ON DELETE SET NULL,
  auxiliar2_id BIGINT REFERENCES cirurgioes(id) ON DELETE SET NULL,
  observacoes TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW())
);

-- Se tabela cirurgias já existe, adicionar coluna numero_mes:
-- ALTER TABLE cirurgias ADD COLUMN numero_mes INTEGER NOT NULL DEFAULT 1;

-- Habilitar RLS
ALTER TABLE cirurgioes ENABLE ROW LEVEL SECURITY;
ALTER TABLE cirurgias ENABLE ROW LEVEL SECURITY;

-- Políticas (acesso público)
CREATE POLICY "Permitir tudo" ON cirurgioes FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Permitir tudo" ON cirurgias FOR ALL USING (true) WITH CHECK (true);
